    8  /** §3 server → client. `{ id, type, ...payload }`; `id` is unique per event. */
    9  
   10  const event = <T extends string, S extends z.ZodRawShape>(type: T, shape: S) =>
   11    z.object({ id: z.string(), type: z.literal(type), ...shape });

   19    ms: z.number().int().nonnegative(),
   20  });

  192  export const Event = z.discriminatedUnion("type", [
  193    SessionStateEvent, SnapshotTaken, TurnStart, Token,
  194    ToolStart, ToolOutput, ToolEnd,
  195    PermissionRequest, AskRequest,
  196    JobStart, JobOutput, JobEnd,
  197    ShowFiles, MentionAdd, FsChanged, GitChanged,
  198    Toast, CompactDone, CompactPlanned, WorkspaceEvent, ErrorEvent,
  199    FsListed, FsTreeListed, FoldersListed, FsContent, FsWrote, GitResult, SessionList, SessionEvents, ConfigEvent,
  200  ]);
  201  export type Event = z.infer<typeof Event>;
  202  export type EventType = Event["type"];
  203  export type EventOf<T extends EventType> = Extract<Event, { type: T }>;
